create database bdCellScale;
use bdCellScale;

create table usuario(
  id int primary key auto_increment,
  nombre varchar(100),
  contrasena varchar(100)
);

create table configuracion(
  id int primary key auto_increment,
  calibracion varchar(100) 
);

create table balanza(
  id int primary key auto_increment,
  COM varchar(100)
);

create table celda_peso(
  id int primary key auto_increment,
  nombre_celda varchar(50),
  valor_peso decimal(10,2),
  fecha_registro datetime 
);
